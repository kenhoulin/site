\documentclass[11pt, letterpaper]{article}

\usepackage[margin=1in]{geometry}
\usepackage{XCharter}            % Charter serif, works with pdflatex
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage[dvipsnames,svgnames,x11names]{xcolor}
\usepackage{hyperref}
\usepackage{microtype}
\usepackage{enumitem}
\usepackage{titlesec}
\usepackage{parskip}
\usepackage{longtable}
\usepackage{booktabs}

% Pandoc emits \tightlist in compact itemize lists.
\providecommand{\tightlist}{\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

% Pandoc 3 emits \begin{longtable*}; alias it to longtable.
\makeatletter
\expandafter\let\csname longtable*\endcsname\longtable
\expandafter\let\csname endlongtable*\endcsname\endlongtable
\makeatother

\hypersetup{
  colorlinks=true,
  urlcolor=blue!60!black,
  linkcolor=black
}

\titleformat{\section}
  {\large\bfseries}{}{0em}{}[\titlerule]
\titlespacing*{\section}{0pt}{1.2em}{0.6em}

\setlist[itemize]{nosep, left=0pt}

\setcounter{secnumdepth}{-1}  % no section/subsection numbering

% Month-year timestamp ("May 2026") for header
\newcommand{\monthyear}{\ifcase\month\or
  January\or February\or March\or April\or May\or June\or
  July\or August\or September\or October\or November\or December\fi
  \space\number\year}
